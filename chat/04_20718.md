@st.cache_data(ttl=3600)
def load_data():
    response = requests.get(DATA_URL, timeout=30)
    response.raise_for_status()

    # CSV를 UTF-8로 읽기
    try:
        df = pd.read_csv(
            BytesIO(response.content),
            encoding="utf-8-sig"
        )
    except UnicodeDecodeError:
        df = pd.read_csv(
            BytesIO(response.content),
            encoding="cp949"
        )

    df.columns = df.columns.str.strip()

    # 숫자로 변환 가능한 열은 숫자형으로 변환
    for col in df.columns:
        if col != "last_name, first_name":
            df[col] = pd.to_numeric(
                df[col], errors="ignore"
            )

    return df

1. code/main.py에서 코드 수정하기
기존 load_data() 함수를 찾아서 아래 코드로 통째로 교체해 줘.


@st.cache_data(ttl=3600)
def load_data():
    import requests
    from io import BytesIO

    DATA_URL = "https://raw.githubusercontent.com/leeseojin09/baseball/refs/heads/main/data/exit_velocity.csv"

    response = requests.get(DATA_URL, timeout=30)
    response.raise_for_status()

    df = pd.read_csv(
        BytesIO(response.content),
        encoding="utf-8-sig"
    )

    # 열 이름의 불필요한 공백 제거
    df.columns = df.columns.str.strip()

    # 선수 이름을 제외한 나머지 열을 숫자로 변환
    name_col = "last_name, first_name"

    for col in df.columns:
        if col != name_col:
            df[col] = pd.to_numeric(df[col], errors="coerce")

    return df
